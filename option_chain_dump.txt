    54: def _call_with_retry(func, *args, **kwargs):
    55:     """
    56:     Wrapper to retry a synchronous XTS call that might fail due to network issues.
    57:     """
    58:     max_retries = 3
    59:     for attempt in range(max_retries):
    60:         try:
    61:             return func(*args, **kwargs)
    62:         except (UnboundLocalError, Exception) as e:
    63:             logger.warning(f"âš ï¸ Network/API error in {func.__name__}: {e}. Retrying... (Attempt {attempt + 1}/{max_retries})")
    64:             if attempt + 1 == max_retries:
    65:                 logger.error(f"âŒ All {max_retries} retries failed for {func.__name__}.")
    66:                 return {'type': 'error', 'description': f'All retries failed for {func.__name__}'}
    67:             time.sleep(0.1 * (attempt + 1))
    68:     return {'type': 'error', 'description': 'Retry logic failed unexpectedly'}
